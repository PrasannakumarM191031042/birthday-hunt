function doGet(e) { // ========================================================== // JSONP API MODE // ========================================================== // // The GitHub QR scanner calls: // // ?api=1&scan=1&callback=qrCallback_xxxxx // // We return: // // qrCallback_xxxxx({ // success:true, // scan:"1", // clue:"...", // destination:"..." // }); // // ========================================================== const api = e && e.parameter && String(e.parameter.api || "").trim(); if (api === "1") { const scanNumber = e.parameter.scan ? String(e.parameter.scan).trim() : ""; const callback = e.parameter.callback ? String(e.parameter.callback).trim() : ""; let response; try { if (!scanNumber) { throw new Error( "Scan number is missing." ); } const huntData = getHuntData(scanNumber); response = { success: true, scan: scanNumber, clue: huntData.clue, destination: huntData.destination }; } catch (error) { response = { success: false, error: error.message || String(error) }; } const json = JSON.stringify(response); // ======================================================== // JSONP RESPONSE // ======================================================== if (callback) { // Only allow a normal JavaScript callback name. // // This prevents arbitrary JavaScript from being // inserted into the response. if ( !/^[A-Za-z_$][0-9A-Za-z_$]*$/.test( callback ) ) { return ContentService .createTextOutput( "Invalid callback." ) .setMimeType( ContentService.MimeType.TEXT ); } return ContentService .createTextOutput( callback + "(" + json + ");" ) .setMimeType( ContentService.MimeType.JAVASCRIPT ); } // ======================================================== // NORMAL JSON RESPONSE // ======================================================== return ContentService .createTextOutput(json) .setMimeType( ContentService.MimeType.JSON ); } // ========================================================== // NORMAL WEB APP PAGE // ========================================================== const template = HtmlService.createTemplateFromFile( "Index" ); const scanNumber = e && e.parameter && e.parameter.scan ? String( e.parameter.scan ).trim() : ""; const scannerMode = !scanNumber || ( e && e.parameter && String( e.parameter.scanner || "" ).trim() === "1" ); template.scanNumber = scanNumber; template.scannerMode = scannerMode; return template .evaluate() .setTitle("Mystery") .setXFrameOptionsMode( HtmlService.XFrameOptionsMode.ALLOWALL ); }
